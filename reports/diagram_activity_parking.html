Диаграмма деятельности (вертикальная)

Подсистема «Отображение свободных и занятых мест». Для Word: Ctrl+P → «Сохранить как PDF» или скриншот Win+Shift+S. Масштаб в браузере: Ctrl + колёсико. 

flowchart TB
  start(("Старт"))

  load["Загрузка данных подсистемы:\nрезультаты детекции с камеры,\nактивные бронирования, slot_overrides"]

  check{"Данные загружены\nкорректно?"}
  err["Сообщение об ошибке,\nповтор запроса к API"]

  show["Отображение:\nполигоны свободных/занятых на кадре камеры,\nкарта мест, счётчики"]

  scenario{"Сценарий"}

  %% --- ветка 1: мониторинг (как «рельеф») ---
  c1["Получение кадра с камеры"]
  c2["Детекция YOLOv8-seg на сервере"]
  c3["Классификация:\nslot_free / slot_occupied"]
  c4["Отрисовка полигонов на overlay\n(зелёный — свободно, красный — занято)"]
  c5["Обновление счётчиков\nсвободно / занято / всего"]

  %% --- ветка 2: карта мест ---
  m1["Показать карту мест и статусы слотов"]
  m2{"Действия\nпользователя?"}
  m_view["Просмотр свободных\nи занятых мест"]
  m_pick["Выбор свободного места"]
  m_check["Проверка bookable(s)"]

  %% --- ветка 3: статусы (как «дефекты») ---
  d1["Показать карту с учётом\nручных статусов"]
  d2{"Изменение\nстатуса места?"}
  d3{"Роль:\nадминистратор?"}
  d_edit["Установка статуса:\nзакрыто / ремонт / резерв"]
  d_save["Сохранение в БД\n(slot_overrides)"]
  d_skip["Только просмотр"]

  %% --- ветка 4: сводка ---
  s1["Агрегация: N свободных,\nзанятых, забронированных"]
  s2["Расчёт индикатора загрузки\n(светофор)"]
  s3["Вывод сводки пользователю"]

  upd["Обновление карты мест,\noverlay камеры и сводных показателей"]

  cont{"Продолжить\nработу?"}
  endnode(("Завершение"))

  %% основной поток сверху вниз
  start --> load --> check
  check -->|нет| err --> load
  check -->|да| show --> scenario

  %% подписи веток у ромба «Сценарий»
  scenario -->|мониторинг\nс камеры| c1
  scenario -->|карта мест| m1
  scenario -->|статусы мест| d1
  scenario -->|сводка| s1

  c1 --> c2 --> c3 --> c4 --> c5 --> upd

  m1 --> m2
  m2 -->|просмотр| m_view --> upd
  m2 -->|бронирование| m_pick --> m_check --> upd

  d1 --> d2
  d2 -->|нет| d_skip --> upd
  d2 -->|да| d3
  d3 -->|да| d_edit --> d_save --> upd
  d3 -->|нет| d_skip

  s1 --> s2 --> s3 --> upd

  upd --> cont
  cont -->|да| scenario
  cont -->|нет| endnode

  classDef action fill:#ffffff,stroke:#333333,stroke-width:1px
  classDef decision fill:#fff8e6,stroke:#b8860b,stroke-width:2px

  class start,endnode startEnd
  class load,err,show,c1,c2,c3,c4,c5,m1,m_view,m_pick,m_check,d1,d_edit,d_save,d_skip,s1,s2,s3,upd action
  class check,scenario,m2,d2,d3,cont decision
